Create migration for `sidebar_items` table if not exists:

```php
Schema::create('sidebar_items', function (Blueprint $table) {
    $table->increments('id');
    $table->string('key')->unique();
    $table->string('label_en');
    $table->string('label_ar');
    $table->string('icon')->nullable();
    $table->string('route_name')->nullable();
    $table->string('url')->nullable();
    $table->string('permission')->nullable();
    $table->string('module');
    $table->integer('parent_id')->nullable()->unsigned();
    $table->integer('order')->default(0);
    $table->boolean('is_visible')->default(true);
    $table->boolean('is_beta')->default(false);
    $table->boolean('is_external')->default(false);
    $table->string('device_visibility')->default('desktop,tablet,mobile');
    $table->string('badge_count_query')->nullable();
    $table->text('active_route_patterns')->nullable();
    $table->timestamps();
    
    $table->foreign('parent_id')->references('id')->on('sidebar_items')->onDelete('set null');
    $table->index(['module', 'order']);
    $table->index('permission');
    $table->index('is_visible');
});
```

### Phase 2: Sidebar Service Class
Create `app/Services/SidebarService.php`:

```php
class SidebarService
{
    public function getMenuForUser($user)
    {
        $userPermissions = $user->getAllPermissions()->pluck('name')->toArray();
        
        return SidebarItem::with('children')
            ->whereNull('parent_id')
            ->where('is_visible', true)
            ->orderBy('order')
            ->get()
            ->filter(function ($item) use ($userPermissions) {
                if (empty($item->permission)) return true;
                return in_array($item->permission, $userPermissions) || 
                       $user->hasRole(['Super Admin', 'Hotel Owner']);
            })
            ->map(function ($item) use ($userPermissions) {
                $item->children = $item->children
                    ->where('is_visible', true)
                    ->filter(function ($child) use ($userPermissions) {
                        if (empty($child->permission)) return true;
                        return in_array($child->permission, $userPermissions) || 
                               $user->hasRole(['Super Admin', 'Hotel Owner']);
                    })
                    ->values();
                return $item;
            });
    }
    
    public function getBadgeCount($item, $teamId)
    {
        if (empty($item->badge_count_query)) return null;
        // Execute badge count query with team scope
        return DB::table(explode('.', $item->badge_count_query)[0])
            ->where('team_id', $teamId)
            ->count();
    }
}
```
